> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter à synoRight une IHM Swing (configuration, droits d'un dossier, synchronisation), un lanceur sans console et un menu contextuel natif Windows 11 (DLL `IExplorerCommand` + MSIX sparse), le menu classique restant disponible.

package sk.synoAdmin.ui;

import java.util.List;

/**
 * Saisies et messages destinés à l'utilisateur : console (CLI) ou dialogues (IHM).
 * Une saisie annulée dans l'IHM lève une UserException("Saisie annulée.").
 */
public interface Interaction {

	/** Demande une valeur non vide. */
	String promptLine(String label);

	/** Demande une valeur, une saisie vide conservant la valeur par défaut (si non nulle). */
	String promptLine(String label, String defaultValue);

	/** Fait choisir un ou plusieurs éléments ; renvoie les index (base 0), sans doublon. */
	List<Integer> promptChoices(String label, List<String> items);

	/** Demande une valeur sensible non vide (saisie masquée). */
	String promptSecret(String label);

	/** Demande une valeur sensible ; saisie vide : null (conserver la valeur actuelle). */
	String promptSecretOrKeep(String label);

	/** Pose une question fermée oui/non. */
	boolean promptBool(String label);

	/** Affiche une information. */
	void info(String message);

	/**
	 * Affiche une erreur.
	 *
	 * @param detail cause à détailler (stack trace), null pour le message seul
	 */
	void error(String message, Throwable detail);

	/** true pour l'IHM (dialogues), false pour la console. */
	boolean isGraphical();
}

```java
package sk.synoAdmin.engine;

import java.util.List;
import java.util.Map;

import sk.synoAdmin.engine.DeltaResolver.GrpDelta;

/**
 * Modifications à appliquer pour synchroniser fichiers et Synology Photos dans un sens.
 *
 * @param deltas par groupe géré (ordre de la configuration), sans les groupes déjà synchronisés
 * @param folderCount nombre de dossiers comparés
 * @param unindexedFolders dossiers du partage pas encore indexés par Synology Photos (ignorés)
 * @param unknownFolders dossiers connus de Synology Photos mais absents du partage (ignorés)
 */
public record SyncPlan(Direction direction, Map<Group, GrpDelta> deltas, int folderCount, List<String> unindexedFolders, List<String> unknownFolders) {

	/** PUSH : fichiers → Synology Photos ; PULL : Synology Photos → fichiers. */
	public enum Direction {
		PUSH, PULL
	}

	public boolean isEmpty() {
		return deltas.isEmpty();
	}

	public int changeCount() {
		return deltas.values().stream().mapToInt(d -> d.getToAdd().size() + d.getToRemove().size()).sum();
	}
}
```

package sk.synoAdmin.engine;

import java.util.List;

/**
 * Droits d'un dossier pour chaque groupe géré, côté Synology Photos et côté fichiers.
 */
public record FolderRights(String folderName, List<GroupRight> rights) {

	public record GroupRight(Group group, boolean inPhotos, boolean inFile) {
		public boolean hasGap() {
			return inPhotos != inFile;
		}
	}

	public boolean hasGap() {
		return rights.stream().anyMatch(GroupRight::hasGap);
	}
}

		for (final Map.Entry<Group, GrpDelta> entry : plan.deltas().entrySet()) {
			final Group group = entry.getKey();
			final GrpDelta delta = entry.getValue();
			if (plan.direction() == SyncPlan.Direction.PUSH) {
				synoClient.setAccessRight(group.getId(), delta.getToAdd(), delta.getToRemove());
				count += delta.getToAdd().size() + delta.getToRemove().size();
			} else {
				for (final Folder folder : delta.getToAdd()) {
					count += doCreateRightFile(group, folder) ? 1 : 0;
				}
				for (final Folder folder : delta.getToRemove()) {
					count += doRemoveRightFile(group, folder) ? 1 : 0;
				}
			}
		}
		return count;
	}

	public FolderRights folderRights(final String folder) {
		final Folder resolved = resolveFolder(folder);
		final List<FolderRights.GroupRight> rights = pref.getGroups().stream()
				.map(g -> new FolderRights.GroupRight(g, resolved.getDbGroups().contains(g), resolved.getFsGroups().contains(g)))
				.toList();
		return new FolderRights(resolved.getName(), rights);
	}

	/**
	 * Donne le droit aux groupes voulus et le retire aux autres groupes gérés, côté Photos et fichiers.
	 *
	 * @return description des modifications effectuées (vide si le dossier était déjà conforme)
	 */
	public List<String> setFolderRights(final String folder, final Set<Group> wanted) {
		final Folder resolved = resolveFolder(folder);
		final List<String> messages = new ArrayList<>();
		for (final Group group : pref.getGroups()) {
			final boolean inDb = resolved.getDbGroups().contains(group);
			final boolean inFs = resolved.getFsGroups().contains(group);
			if (wanted.contains(group) && !(inDb && inFs)) {
				messages.add(addRight(resolved, group));
			} else if (!wanted.contains(group) && (inDb || inFs)) {
				messages.add(removeRight(resolved, group));
			}
		}
		return messages;
	}

	/** Id Synology Photos du dossier. */
	public int folderId(final String folder) {
		return resolveFolder(folder).getId();
	}

package sk.synoAdmin.menu;

import java.util.List;
import java.util.stream.Collectors;

/**
 * Ligne de commande Windows (règles de CommandLineToArgvW) : guillemets et backslashs.
 */
public final class CommandLines {

	private CommandLines() {
		// utility
	}

	public static String quote(final String arg) {
		final boolean needsQuotes = arg.isEmpty() || arg.equals(MenuDefinition.PATH)
				|| arg.chars().anyMatch(c -> c == ' ' || c == '\t' || c == '"');
		if (!needsQuotes) {
			return arg;
		}
		final StringBuilder quoted = new StringBuilder("\"");
		int backslashes = 0;
		for (final char c : arg.toCharArray()) {
			if (c == '\\') {
				backslashes++;
				continue;
			}
			if (c == '"') {
				// backslashs précédant un guillemet doublés, guillemet échappé
				quoted.append("\\".repeat(backslashes * 2 + 1)).append('"');
			} else {
				quoted.append("\\".repeat(backslashes)).append(c);
			}
			backslashes = 0;
		}
		// backslashs finaux doublés : ils précèdent le guillemet fermant
		quoted.append("\\".repeat(backslashes * 2)).append('"');
		return quoted.toString();
	}

	public static String join(final List<String> args) {
		return args.stream().map(CommandLines::quote).collect(Collectors.joining(" "));
	}
}

	/**
	 * @param key nom de clé de registre (ordre d'affichage)
	 * @param separatorBefore séparateur avant l'entrée (première entrée d'un bloc)
	 */
	public record MenuItem(String key, String title, List<String> args, boolean separatorBefore) {
	}

	public static List<MenuItem> build(final List<MenuEntry> entries, final List<Group> groups) {
		final List<MenuItem> items = new ArrayList<>();
		// blocs : dossier / synchro / groupes / configuration
		addBlock(items, entries, groups, MenuEntry.RIGHTS, MenuEntry.OPEN);
		addBlock(items, entries, groups, MenuEntry.STATUS, MenuEntry.PUSH, MenuEntry.PULL);
		addBlock(items, entries, groups, MenuEntry.GROUPS);
		addBlock(items, entries, groups, MenuEntry.CONFIG);
		return items;
	}

	private static void addBlock(final List<MenuItem> items, final List<MenuEntry> entries, final List<Group> groups, final MenuEntry... block) {
		boolean first = true;
		for (final MenuEntry entry : block) {
			if (!entries.contains(entry)) {
				continue;
			}
			for (final String[] item : itemsOf(entry, groups)) {
				final boolean separator = first && !items.isEmpty();
				final String key = String.format("%02d%s", items.size() + 1, item[0]);
				items.add(new MenuItem(key, item[1], List.of(item).subList(2, item.length), separator));
				first = false;
			}
		}
	}

	// {suffixe de clé, titre, arguments...}
	private static List<String[]> itemsOf(final MenuEntry entry, final List<Group> groups) {
		return switch (entry) {
			case RIGHTS -> List.<String[]>of(new String[] { "rights", entry.label(), "rights", PATH });
			case OPEN -> List.<String[]>of(new String[] { "open", entry.label(), "open", PATH });
			case STATUS -> List.<String[]>of(new String[] { "status", entry.label(), "sync" });
			case PUSH -> List.<String[]>of(new String[] { "push", entry.label(), "sync", "--push" });
			case PULL -> List.<String[]>of(new String[] { "pull", entry.label(), "sync", "--pull" });
			case CONFIG -> List.<String[]>of(new String[] { "config", entry.label(), "config" });
			case GROUPS -> groups.stream()
					.<String[]>mapMulti((g, out) -> {
						out.accept(new String[] { "add" + g.getId(), "Ajouter " + g.getName(), "grant", g.getName(), PATH });
						out.accept(new String[] { "del" + g.getId(), "Retirer " + g.getName(), "revoke", g.getName(), PATH });
					})
					.toList();
		};
	}

package sk.synoAdmin.menu;

/**
 * Menu du nouveau menu contextuel de Windows 11 : DLL synoRightMenu.dll déclarée par le
 * package sparse synoRight.msix (signé par un certificat auto-signé), configurée par
 * HKCU\Software\synoRight\Menu que la DLL relit à chaque clic droit.
 */
public final class ModernMenuInstaller {

	public static final String PACKAGE_NAME = "synoRight";
	static final String MENU_KEY = "HKEY_CURRENT_USER\\Software\\synoRight\\Menu";
	static final String DLL = "synoRightMenu.dll";
	static final String MSIX = "synoRight.msix";
	static final String CER = "synoRight.cer";
	private static final int WINDOWS_11_BUILD = 22000;

	public record Availability(boolean available, String reason) {
	}

	private ModernMenuInstaller() {
		// utility
	}

	public static Availability availability() {
		if (!ProcessRunner.isWindows()) {
			return new Availability(false, "disponible seulement sous Windows 11");
		}
		if (windowsBuild() < WINDOWS_11_BUILD) {
			return new Availability(false, "nécessite Windows 11");
		}
		final Optional<Path> appDir = Launchers.appDir();
		if (appDir.isEmpty()) {
			return new Availability(false, "nécessite l'application installée (synoRightGui.exe), pas le jar seul");
		}
		for (final String file : List.of(DLL, MSIX, CER)) {
			if (!Files.isRegularFile(appDir.get().resolve(file))) {
				return new Availability(false, "fichier " + file + " absent de " + appDir.get() + " (application construite sans le menu Windows 11)");
			}
		}
		return new Availability(true, null);
	}

	private static int windowsBuild() {
		final ProcessRunner.Result result = ProcessRunner.run(List.of("reg", "query",
				"HKLM\\SOFTWARE\\Microsoft\\Windows NT\\CurrentVersion", "/v", "CurrentBuildNumber"));
		final Matcher matcher = Pattern.compile("CurrentBuildNumber\\s+REG_SZ\\s+(\\d+)").matcher(result.output());
		return matcher.find() ? Integer.parseInt(matcher.group(1)) : 0;
	}

	public static void install(final List<MenuItem> items, final Path configDir, final String photoFolder) {
		final Availability availability = availability();
		if (!availability.available()) {
			throw new UserException("Menu Windows 11 indisponible : " + availability.reason() + ".");
		}
		final Path appDir = Launchers.appDir().orElseThrow();
		trustCertificate(appDir.resolve(CER));

		// même version et contenu différent refusés par Add-AppxPackage : on retire l'ancien package
		if (isInstalled()) {
			powershell("Get-AppxPackage -Name " + PACKAGE_NAME + " | Remove-AppxPackage");
		}
		final ProcessRunner.Result added = powershell("Add-AppxPackage -Path " + psQuote(appDir.resolve(MSIX).toString())
				+ " -ExternalLocation " + psQuote(appDir.toString()));
		if (!added.ok()) {
			throw new UserException("Installation du package du menu Windows 11 en échec : " + added.output());
		}
		refreshEntries(items, configDir, photoFolder);
	}

	public static void refreshEntries(final List<MenuItem> items, final Path configDir, final String photoFolder) {
		final String launcher = Launchers.appDir().orElseThrow().resolve(Launchers.GUI_EXE).toString();
		ClassicMenuInstaller.importRegFile(buildRegFileContent(items, launcher, configDir, photoFolder));
	}

	public static void uninstall() {
		powershell("Get-AppxPackage -Name " + PACKAGE_NAME + " | Remove-AppxPackage");
		ProcessRunner.run(List.of("reg", "delete", "HKCU\\Software\\synoRight", "/f"));
	}

	public static boolean isInstalled() {
		return ProcessRunner.isWindows()
				&& powershell("if (Get-AppxPackage -Name " + PACKAGE_NAME + ") { 'installe' }").output().contains("installe");
	}

	static String buildRegFileContent(final List<MenuItem> items, final String launcherExe, final Path configDir, final String photoFolder) {
		final StringBuilder reg = new StringBuilder("Windows Registry Editor Version 5.00\r\n");
		// suppression préalable : entrées obsolètes
		reg.append("\r\n[-").append(MENU_KEY).append("]\r\n");
		reg.append("\r\n[").append(MENU_KEY).append("]\r\n")
				.append("\"Launcher\"=\"").append(escape(launcherExe)).append("\"\r\n")
				.append("\"ConfigDir\"=\"").append(escape(configDir.toString())).append("\"\r\n")
				.append("\"PhotoFolder\"=\"").append(escape(photoFolder)).append("\"\r\n");
		int index = 1;
		for (final MenuItem item : items) {
			final List<String> args = new ArrayList<>(List.of("-c", configDir.toString()));
			args.addAll(item.args());
			reg.append("\r\n[").append(MENU_KEY).append("\\Entries\\").append(String.format("%02d", index++)).append("]\r\n")
					.append("\"Title\"=\"").append(escape(item.title())).append("\"\r\n")
					.append("\"Args\"=\"").append(escape(CommandLines.join(args))).append("\"\r\n")
					.append("\"SeparatorBefore\"=dword:0000000").append(item.separatorBefore() ? "1" : "0").append("\r\n");
		}
		return reg.toString();
	}

	// échappement du format .reg : backslashs et guillemets
	private static String escape(final String value) {
		return value.replace("\\", "\\\\").replace("\"", "\\\"");
	}

	/** Chaîne PowerShell entre apostrophes. */
	static String psQuote(final String value) {
		return "'" + value.replace("'", "''") + "'";
	}

	private static ProcessRunner.Result powershell(final String command) {
		return ProcessRunner.run(List.of("powershell", "-NoProfile", "-NonInteractive", "-Command", command));
	}

	/** Approuve le certificat du package (magasin machine TrustedPeople) : une élévation UAC la première fois. */
	private static void trustCertificate(final Path cer) {
		final String thumbprint = thumbprint(cer);
		final String check = "Test-Path " + psQuote("Cert:\\LocalMachine\\TrustedPeople\\" + thumbprint);
		if (powershell(check).output().contains("True")) {
			return;
		}
		ConsoleUtil.info("Windows va demander une autorisation administrateur (une seule fois) pour approuver le certificat du menu Windows 11.");
		final String importCommand = "Import-Certificate -FilePath " + psQuote(cer.toString()) + " -CertStoreLocation Cert:\\LocalMachine\\TrustedPeople";
		powershell("Start-Process powershell -Verb RunAs -Wait -WindowStyle Hidden -ArgumentList '-NoProfile','-Command',"
				+ psQuote(importCommand));
		if (!powershell(check).output().contains("True")) {
			throw new UserException("Certificat du menu Windows 11 non approuvé (autorisation refusée ?) : menu non installé. Le menu classique reste disponible.");
		}
	}

	// empreinte SHA-1 (thumbprint Windows) du certificat
	static String thumbprint(final Path cer) {
		try (InputStream in = Files.newInputStream(cer)) {
			final Certificate certificate = CertificateFactory.getInstance("X.509").generateCertificate(in);
			return HexFormat.of().withUpperCase().formatHex(MessageDigest.getInstance("SHA-1").digest(certificate.getEncoded()));
		} catch (final IOException | GeneralSecurityException e) {
			throw new UserException("Certificat du menu illisible : " + cer, e);
		}
	}
}

package sk.synoAdmin.menu;

/**
 * Installation des menus contextuels selon la configuration (entrées, cible).
 */
public final class MenuService {

	public record Status(boolean classic, boolean modern) {
		public boolean any() {
			return classic || modern;
		}
	}

	private MenuService() {
		// utility
	}

	/** Cible configurée, sinon Windows 11 si disponible, sinon classique. */
	public static MenuTarget effectiveTarget(final Preferences pref) {
		if (pref.getMenuTarget() != null) {
			return pref.getMenuTarget();
		}
		return ModernMenuInstaller.availability().available() ? MenuTarget.MODERN : MenuTarget.CLASSIC;
	}

	/** Installe le(s) menu(s) de la cible et retire l'autre ; renvoie un message de résultat. */
	public static String install(final Preferences pref, final MenuTarget target) {
		final List<MenuItem> items = MenuDefinition.build(pref.getMenuEntries(), pref.getGroups());
		final boolean modern = target != MenuTarget.CLASSIC;
		final boolean classic = target != MenuTarget.MODERN;

		if (modern) {
			ModernMenuInstaller.install(items, pref.getConfigDir(), pref.getNasFolder());
		} else if (ModernMenuInstaller.isInstalled()) {
			ModernMenuInstaller.uninstall();
		}
		if (classic) {
			ClassicMenuInstaller.install(items, pref.getNasFolder(), pref.getConfigDir());
		} else {
			ClassicMenuInstaller.uninstall();
		}
		return "Menu contextuel installé pour l'utilisateur courant ("
				+ (modern && classic ? "Windows 11 et classique" : modern ? "Windows 11" : "classique") + ", "
				+ items.size() + " entrée(s)).";
	}

	public static void uninstall() {
		if (ModernMenuInstaller.isInstalled()) {
			ModernMenuInstaller.uninstall();
		}
		ClassicMenuInstaller.uninstall();
	}

	public static Status status() {
		return new Status(ClassicMenuInstaller.isInstalled(), ModernMenuInstaller.isInstalled());
	}

	/** Met à jour les menus installés après un changement de groupes ou d'entrées (sans réinstaller le package). */
	public static void refresh(final Preferences pref) {
		final List<MenuItem> items = MenuDefinition.build(pref.getMenuEntries(), pref.getGroups());
		if (ClassicMenuInstaller.isInstalled()) {
			ClassicMenuInstaller.install(items, pref.getNasFolder(), pref.getConfigDir());
		}
		if (ModernMenuInstaller.isInstalled()) {
			ModernMenuInstaller.refreshEntries(items, pref.getConfigDir(), pref.getNasFolder());
		}
	}
}

// Aucune logique métier : les entrées sont lues dans HKCU\Software\synoRight\Menu
// (écrit par synoRight) à chaque ouverture du menu, et lancent synoRightGui.exe.

constexpr wchar_t ENTRIES_KEY[] = L"Software\\synoRight\\Menu\\Entries";
constexpr wchar_t ROOT_TITLE[] = L"Droits Synology Photos";

struct Entry {
    std::wstring title;
    std::wstring args;
    bool separatorBefore = false;

    std::wstring launcher;
    std::wstring photoFolder;
    std::vector<Entry> entries;

bool ReadString(HKEY root, const wchar_t* subKey, const wchar_t* name, std::wstring& out) {
    DWORD size = 0;
    if (RegGetValueW(root, subKey, name, RRF_RT_REG_SZ, nullptr, nullptr, &size) != ERROR_SUCCESS || size == 0) {
        return false;
    }
    std::wstring buffer(size / sizeof(wchar_t), L'\0');
    if (RegGetValueW(root, subKey, name, RRF_RT_REG_SZ, nullptr, buffer.data(), &size) != ERROR_SUCCESS) {
        return false;
    }
    buffer.resize(wcsnlen(buffer.c_str(), buffer.size()));
    out = std::move(buffer);
    return !out.empty();
}

DWORD ReadDword(HKEY root, const wchar_t* subKey, const wchar_t* name) {
    DWORD value = 0;
    DWORD size = sizeof(value);
    if (RegGetValueW(root, subKey, name, RRF_RT_REG_DWORD, nullptr, &value, &size) != ERROR_SUCCESS) {
        return 0;
    }
    return value;
}

bool LoadConfig(Config& config) {
    if (!ReadString(HKEY_CURRENT_USER, MENU_KEY, L"Launcher", config.launcher)
        || !ReadString(HKEY_CURRENT_USER, MENU_KEY, L"PhotoFolder", config.photoFolder)) {
        return false;
    }
    HKEY entries = nullptr;
    if (RegOpenKeyExW(HKEY_CURRENT_USER, ENTRIES_KEY, 0, KEY_READ, &entries) != ERROR_SUCCESS) {
        return false;
    }
    std::vector<std::wstring> names;
    for (DWORD index = 0;; ++index) {
        wchar_t name[64];
        DWORD length = ARRAYSIZE(name);
        if (RegEnumKeyExW(entries, index, name, &length, nullptr, nullptr, nullptr, nullptr) != ERROR_SUCCESS) {
            break;
        }
        names.emplace_back(name, length);
    }
    std::sort(names.begin(), names.end()); // 01, 02... : ordre d'affichage
    for (const auto& name : names) {
        Entry entry;
        if (ReadString(entries, name.c_str(), L"Title", entry.title) && ReadString(entries, name.c_str(), L"Args", entry.args)) {
            entry.separatorBefore = ReadDword(entries, name.c_str(), L"SeparatorBefore") != 0;
            config.entries.push_back(std::move(entry));
        }
    }
    RegCloseKey(entries);
    return !config.entries.empty();
}

std::wstring TrimSlashes(std::wstring path) {
    while (path.size() > 3 && (path.back() == L'\\' || path.back() == L'/')) {
        path.pop_back();
    }
    return path;
}

// dossier égal au dossier photos ou en dessous, sans tenir compte de la casse
bool IsUnder(const std::wstring& path, const std::wstring& root) {
    const std::wstring p = TrimSlashes(path);
    const std::wstring r = TrimSlashes(root);
    if (p.size() < r.size()
        || CompareStringOrdinal(p.c_str(), static_cast<int>(r.size()), r.c_str(), static_cast<int>(r.size()), TRUE) != CSTR_EQUAL) {
        return false;
    }
    return p.size() == r.size() || r.back() == L'\\' || p[r.size()] == L'\\';
}

bool FirstPath(IShellItemArray* items, std::wstring& path) {
    DWORD count = 0;
    if (items == nullptr || FAILED(items->GetCount(&count)) || count == 0) {
        return false;
    }
    ComPtr<IShellItem> item;
    if (FAILED(items->GetItemAt(0, &item))) {
        return false;
    }
    PWSTR name = nullptr;
    if (FAILED(item->GetDisplayName(SIGDN_FILESYSPATH, &name))) {
        return false;
    }
    path = name;
    CoTaskMemFree(name);
    return true;
}

// remplace %V par le dossier ; "P:\" entre guillemets : le \ final échapperait le guillemet
std::wstring ExpandArgs(const std::wstring& args, std::wstring path) {
    if (!path.empty() && path.back() == L'\\') {
        path.push_back(L'\\');
    }
    std::wstring result = args;
    for (size_t pos = 0; (pos = result.find(L"%V", pos)) != std::wstring::npos; pos += path.size()) {
        result.replace(pos, 2, path);
    }
    return result;
}

class SubCommand final : public RuntimeClass<RuntimeClassFlags<ClassicCom>, IExplorerCommand> {
public:

        *name = nullptr;
        return separator_ ? E_NOTIMPL : SHStrDupW(title_.c_str(), name);
    }
    IFACEMETHODIMP GetIcon(IShellItemArray*, LPWSTR* icon) override {
        *icon = nullptr;
        return E_NOTIMPL;
    }
    IFACEMETHODIMP GetToolTip(IShellItemArray*, LPWSTR* tip) override {
        *tip = nullptr;
        return E_NOTIMPL;
    }
    IFACEMETHODIMP GetCanonicalName(GUID* guid) override {
        *guid = GUID_NULL;
        return S_OK;
    }
    IFACEMETHODIMP GetState(IShellItemArray*, BOOL, EXPCMDSTATE* state) override {
        *state = ECS_ENABLED;
        return S_OK;
    }
    IFACEMETHODIMP Invoke(IShellItemArray* items, IBindCtx*) override {
        std::wstring path;
        if (!FirstPath(items, path)) {
            return S_OK;
        }
        std::wstring commandLine = L"\"" + launcher_ + L"\" " + ExpandArgs(args_, path);
        STARTUPINFOW startup{};
        startup.cb = sizeof(startup);
        PROCESS_INFORMATION process{};
        if (CreateProcessW(launcher_.c_str(), commandLine.data(), nullptr, nullptr, FALSE, 0, nullptr, nullptr, &startup, &process)) {
            CloseHandle(process.hThread);
            CloseHandle(process.hProcess);
        }
        return S_OK;

        *flags = separator_ ? ECF_ISSEPARATOR : ECF_DEFAULT;
        return S_OK;
    }

        *commands = nullptr;
        return E_NOTIMPL;
    }

private:
    std::wstring title_;
    std::wstring launcher_;
    std::wstring args_;
    bool separator_;
};

class CommandEnum final : public RuntimeClass<RuntimeClassFlags<ClassicCom>, IEnumExplorerCommand> {
public:
    explicit CommandEnum(std::vector<ComPtr<IExplorerCommand>> commands) : commands_(std::move(commands)) {}

    IFACEMETHODIMP Next(ULONG count, IExplorerCommand** out, ULONG* fetched) override {
        ULONG done = 0;
        for (; done < count && index_ < commands_.size(); ++done, ++index_) {
            commands_[index_].CopyTo(&out[done]);
        }
        if (fetched != nullptr) {
            *fetched = done;
        }
        return done == count ? S_OK : S_FALSE;
    }
    IFACEMETHODIMP Skip(ULONG count) override {
        index_ = std::min<size_t>(index_ + count, commands_.size());
        return S_OK;
    }
    IFACEMETHODIMP Reset() override {
        index_ = 0;
        return S_OK;
    }
    IFACEMETHODIMP Clone(IEnumExplorerCommand** out) override {
        *out = nullptr;
        return E_NOTIMPL;
    }

private:
    std::vector<ComPtr<IExplorerCommand>> commands_;
    size_t index_ = 0;
};

	/** Exécute la ligne de commande et renvoie le code de sortie (sans quitter la JVM). */
	static int run(final String[] args) {
		final long start = System.nanoTime();
		final int exitCode;
		if (isLegacySyntax(args)) {
			// messages actuels inchangés
			exitCode = 2;
		} else {
			// construction actuelle de CommandLine, puis :
			exitCode = commandLine.execute(args);
		}
		if (LogUtil.isVerbose()) {
			System.out.println("Total time : " + (System.nanoTime() - start) / 1_000_000 + " ms");
		}
		return exitCode;
	}

	// erreur prévue : message seul ; imprévue : message, détail en mode verbeux (toujours dans l'IHM)
	static int handleError(final Exception e) {
		final Interaction ui = ConsoleUtil.interaction();
		if (e instanceof UserException) {
			ui.error(e.getMessage(), LogUtil.isVerbose() ? e.getCause() : null);
		} else if (ui.isGraphical()) {
			ui.error("Erreur inattendue : " + e, e);
		} else {
			ui.error("Erreur inattendue : " + e, LogUtil.isVerbose() ? e : null);
			if (!LogUtil.isVerbose()) {
				System.err.println("(relancer avec -v pour le détail)");
			}
		}
		return 1;
	}

public final class Gui {

	/** Thème FlatLaf (clair ou sombre comme Windows) et dialogues pour les saisies. */
	public static void setup() {
		if (GraphicsEnvironment.isHeadless()) {
			throw new UserException("Pas d'affichage graphique disponible : utiliser les commandes en console (synoRight --help).");
		}
		if (windowsDarkTheme()) {

	private static boolean windowsDarkTheme() {
		if (!ProcessRunner.isWindows()) {
			return false;
		}
		final ProcessRunner.Result result = ProcessRunner.run(List.of("reg", "query",
				"HKCU\\Software\\Microsoft\\Windows\\CurrentVersion\\Themes\\Personalize", "/v", "AppsUseLightTheme"));
		return result.ok() && result.output().contains("0x0");
	}

	/** Affiche la fenêtre et attend sa fermeture (dispose). */
	public static void showAndWait(final Window window) {
		final CountDownLatch closed = new CountDownLatch(1);
		window.addWindowListener(new WindowAdapter() {
			@Override
			public void windowClosed(final WindowEvent e) {
				closed.countDown();
			}
		});
		EventQueue.invokeLater(() -> {
			window.setLocationRelativeTo(null);
			window.setVisible(true);
		});
		try {
			closed.await();
		} catch (final InterruptedException e) {
			Thread.currentThread().interrupt();
		}
	}

	/**
	 * Exécute une tâche longue (réseau, disque) hors de l'EDT : curseur d'attente, owner désactivé
	 * pendant l'opération, résultat ou erreur traités sur l'EDT.
	 */
	public static <T> void runAsync(final Component owner, final Supplier<T> task, final Consumer<T> onSuccess) {
		owner.setEnabled(false);
		owner.setCursor(Cursor.getPredefinedCursor(Cursor.WAIT_CURSOR));
		new SwingWorker<T, Void>() {
			@Override
			protected T doInBackground() {
				return task.get();
			}

			@Override
			protected void done() {
				owner.setEnabled(true);
				owner.setCursor(Cursor.getDefaultCursor());
				try {
					onSuccess.accept(get());
				} catch (final ExecutionException e) {
					showError(owner, e.getCause());
				} catch (final InterruptedException e) {
					Thread.currentThread().interrupt();
				}
			}
		}.execute();
	}

	/** Erreur prévue (UserException) : message seul ; sinon message + détail repliable. */
	public static void showError(final Component parent, final Throwable e) {
		if (e instanceof UserException) {
			JOptionPane.showMessageDialog(parent, e.getMessage(), TITLE, JOptionPane.ERROR_MESSAGE);
			return;
		}
		showError(parent, "Erreur inattendue : " + e, e);
	}

	static void showError(final Component parent, final String message, final Throwable detail) {
		if (detail == null) {
			JOptionPane.showMessageDialog(parent, message, TITLE, JOptionPane.ERROR_MESSAGE);
			return;
		}
		final StringWriter trace = new StringWriter();
		detail.printStackTrace(new PrintWriter(trace));
		final JTextArea area = new JTextArea(trace.toString(), 15, 70);
		area.setEditable(false);
		final JScrollPane details = new JScrollPane(area);
		details.setVisible(false);
		final JCheckBox toggle = new JCheckBox("Afficher le détail");
		final JPanel panel = new JPanel(new BorderLayout(0, 8));
		panel.add(new JLabel(message), BorderLayout.NORTH);
		panel.add(toggle, BorderLayout.CENTER);
		panel.add(details, BorderLayout.SOUTH);
		toggle.addActionListener(ev -> {
			details.setVisible(toggle.isSelected());
			SwingUtilities.getWindowAncestor(panel).pack();
		});
		JOptionPane.showMessageDialog(parent, panel, TITLE, JOptionPane.ERROR_MESSAGE);
	}

	public static boolean confirm(final Component parent, final String message) {
		return JOptionPane.showConfirmDialog(parent, message, TITLE, JOptionPane.YES_NO_OPTION) == JOptionPane.YES_OPTION;
	}
}

package sk.synoAdmin.gui;

/**
 * Saisies par dialogues Swing, appelables depuis n'importe quel thread (exécutées sur l'EDT).
 * Une annulation lève UserException("Saisie annulée.").
 */
final class SwingInteraction implements Interaction {

	private static final String CANCELLED = "Saisie annulée.";

	@Override
	public String promptLine(final String label) {
		while (true) {
			final String value = promptLine(label, null);
			if (!value.isEmpty()) {
				return value;
			}
		}
	}

	@Override
	public String promptLine(final String label, final String defaultValue) {
		final String value = onEdt(() -> (String) JOptionPane.showInputDialog(null, label, Gui.TITLE,
				JOptionPane.QUESTION_MESSAGE, null, null, defaultValue));
		if (value == null) {
			throw new UserException(CANCELLED);
		}
		return value.trim().isEmpty() && defaultValue != null ? defaultValue : value.trim();
	}

	@Override
	public List<Integer> promptChoices(final String label, final List<String> items) {
		while (true) {
			final List<Integer> choices = onEdt(() -> {
				final JPanel panel = new JPanel(new GridLayout(0, 1));
				panel.add(new JLabel(label));
				final List<JCheckBox> boxes = items.stream().map(JCheckBox::new).toList();
				boxes.forEach(panel::add);
				if (JOptionPane.showConfirmDialog(null, panel, Gui.TITLE, JOptionPane.OK_CANCEL_OPTION) != JOptionPane.OK_OPTION) {
					return null;
				}
				final List<Integer> selected = new ArrayList<>();
				for (int i = 0; i < boxes.size(); i++) {
					if (boxes.get(i).isSelected()) {
						selected.add(i);
					}
				}
				return selected;
			});
			if (choices == null) {
				throw new UserException(CANCELLED);
			}
			if (!choices.isEmpty()) {
				return choices;
			}
		}
	}

	@Override
	public String promptSecret(final String label) {
		while (true) {
			final String value = readSecret(label);
			if (!value.isEmpty()) {
				return value;
			}
		}
	}

	@Override
	public String promptSecretOrKeep(final String label) {
		final String value = readSecret(label);
		return value.isEmpty() ? null : value;
	}

	private static String readSecret(final String label) {
		final String value = onEdt(() -> {
			final JPasswordField field = new JPasswordField(20);
			final JPanel panel = new JPanel(new BorderLayout(0, 4));
			panel.add(new JLabel(label), BorderLayout.NORTH);
			panel.add(field, BorderLayout.CENTER);
			field.addAncestorListener(new AncestorListener() {
				@Override
				public void ancestorAdded(final AncestorEvent e) {
					field.requestFocusInWindow();
				}
				@Override
				public void ancestorRemoved(final AncestorEvent e) {
					// rien
				}
				@Override
				public void ancestorMoved(final AncestorEvent e) {
					// rien
				}
			});
			final int choice = JOptionPane.showConfirmDialog(null, panel, Gui.TITLE, JOptionPane.OK_CANCEL_OPTION, JOptionPane.QUESTION_MESSAGE);
			return choice == JOptionPane.OK_OPTION ? new String(field.getPassword()) : null;
		});
		if (value == null) {
			throw new UserException(CANCELLED);
		}
		return value.trim();
	}

	@Override
	public boolean promptBool(final String label) {
		return onEdt(() -> JOptionPane.showConfirmDialog(null, label, Gui.TITLE, JOptionPane.YES_NO_OPTION) == JOptionPane.YES_OPTION);
	}

	@Override
	public void info(final String message) {
		onEdt(() -> {
			JOptionPane.showMessageDialog(null, message, Gui.TITLE, JOptionPane.INFORMATION_MESSAGE);
			return null;
		});
	}

	@Override
	public void error(final String message, final Throwable detail) {
		onEdt(() -> {
			Gui.showError(null, message, detail);
			return null;
		});
	}

	@Override
	public boolean isGraphical() {
		return true;
	}

	private static <T> T onEdt(final Supplier<T> dialog) {
		if (EventQueue.isDispatchThread()) {
			return dialog.get();
		}
		final AtomicReference<T> result = new AtomicReference<>();
		try {
			EventQueue.invokeAndWait(() -> result.set(dialog.get()));
		} catch (final InvocationTargetException e) {
			if (e.getCause() instanceof final RuntimeException cause) {
				throw cause;
			}
			throw new RuntimeException(e.getCause());
		} catch (final InterruptedException e) {
			Thread.currentThread().interrupt();
			throw new UserException(CANCELLED);
		}
		return result.get();
	}
}

	static Optional<Set<Integer>> parseSelection(final String input, final int size) {
		final Set<Integer> selection = new LinkedHashSet<>();
		for (final String part : input.split("[,; ]+")) {
			if (part.isBlank()) {
				continue;
			}
			try {
				final int value = Integer.parseInt(part.trim());
				if (value == 0) {
					continue;
				}
				if (value < 1 || value > size) {
					return Optional.empty();
				}
				selection.add(value - 1);
			} catch (final NumberFormatException e) {
				return Optional.empty();
			}
		}
		return Optional.of(selection);
	}
